-- A private division-free expression grammar; unlike PolyExpr it has no power
-- constructor. Its three-variable derivatives are proved structurally below.
private inductive E where

  | var : Nat → E
  | add : E → E → E
  | mul : E → E → E
  | neg : E → E
private instance : Add E := ⟨.add⟩
private instance : Mul E := ⟨.mul⟩
private instance : Neg E := ⟨.neg⟩
private instance (n : Nat) : OfNat E n := ⟨.const (Int.ofNat n)⟩
private def pe : E → PolyExpr
  | .const n => .const n
  | .var n => .var n
  | .add p q => .add (pe p) (pe q)
  | .mul p q => .mul (pe p) (pe q)
  | .neg p => .neg (pe p)
private def exprDerivative (i : Nat) : E → E
  | .const _ => 0
  | .var j => if i=j then 1 else 0
  | .add p q => exprDerivative i p+exprDerivative i q
  | .mul p q => exprDerivative i p*q+p*exprDerivative i q
  | .neg p => -(exprDerivative i p)

theorem cayley_cleared_left_tangent (e : Q') (u : Triple) :
    PaddedQuaternionCoeffEqv 1 (clearedQuotientTangent e u)
      (qscale (cayleyD e) (PolyQuaternion.mul (leftGenerator e u) (cayleyN e))) :=
  paddedQuaternion_components (tangent_scalar_actual e u) (tangent_i_actual e u)
    (tangent_j_actual e u) (tangent_k_actual e u)

private def component (c : Fin 4) (q : PolyQuaternion) : RatPoly :=
  match c.val with | 0 => q.scalar | 1 => q.iPart | 2 => q.jPart | _ => q.kPart
private def qsmul (s : Q') (q : PolyQuaternion) : PolyQuaternion :=
  ⟨polySmul s q.scalar,polySmul s q.iPart,polySmul s q.jPart,polySmul s q.kPart⟩
private def qadd (p q : PolyQuaternion) : PolyQuaternion :=
  ⟨polyAdd p.scalar q.scalar,polyAdd p.iPart q.iPart,
    polyAdd p.jPart q.jPart,polyAdd p.kPart q.kPart⟩

/-- The EXISTING original left-frame basis, with its actual 2e*u_i weights.
coordinate_action false identifies each imaginaryBasis i with the LEFT
original-coordinate action. This definition does not assert an arbitrary-source
frame pullback through the chart. -/
def originalLeftGenerator (e : Q') (u : Triple) : PolyQuaternion :=
  qadd (qsmul (2*e*u.x) (imaginaryBasis 0))
    (qadd (qsmul (2*e*u.y) (imaginaryBasis 1)) (qsmul (2*e*u.z) (imaginaryBasis 2)))

private theorem basis_component (c : Fin 4) (x y z : Q') :
    RatPolyCoeffEqv (polyConst (match c.val with | 0 => 0 | 1 => x | 2 => y | _ => z))
      (polyAdd (polySmul x (component c (imaginaryBasis 0)))
        (polyAdd (polySmul y (component c (imaginaryBasis 1)))
          (polySmul z (component c (imaginaryBasis 2))))) := by
  have hc : c=0 ∨ c=1 ∨ c=2 ∨ c=3 := by simp only [Fin.ext_iff]; omega
  rcases hc with rfl | rfl | rfl | rfl
  · intro weight
    exact poly_identity_of_normal_eq
      (.add (.mul (.const 0) (.var 3)) (.const 0))
      (.add (.mul (.mul (.var 0) (.const 0)) (.var 3))
        (.add (.mul (.mul (.var 1) (.const 0)) (.var 3))
          (.add (.mul (.mul (.var 2) (.const 0)) (.var 3)) (.const 0))))
      (by decide) (fun | 0 => x | 1 => y | 2 => z | _ => weight [])
  · intro weight
    exact poly_identity_of_normal_eq
      (.add (.mul (.var 0) (.var 3)) (.const 0))
      (.add (.mul (.mul (.var 0) (.const 1)) (.var 3))
        (.add (.mul (.mul (.var 1) (.const 0)) (.var 3))
          (.add (.mul (.mul (.var 2) (.const 0)) (.var 3)) (.const 0))))
      (by decide) (fun | 0 => x | 1 => y | 2 => z | _ => weight [])
  · intro weight
    exact poly_identity_of_normal_eq
      (.add (.mul (.var 1) (.var 3)) (.const 0))
      (.add (.mul (.mul (.var 0) (.const 0)) (.var 3))
        (.add (.mul (.mul (.var 1) (.const 1)) (.var 3))
          (.add (.mul (.mul (.var 2) (.const 0)) (.var 3)) (.const 0))))
      (by decide) (fun | 0 => x | 1 => y | 2 => z | _ => weight [])
  · intro weight
    exact poly_identity_of_normal_eq
      (.add (.mul (.var 2) (.var 3)) (.const 0))
      (.add (.mul (.mul (.var 0) (.const 0)) (.var 3))
        (.add (.mul (.mul (.var 1) (.const 0)) (.var 3))
          (.add (.mul (.mul (.var 2) (.const 1)) (.var 3)) (.const 0))))
      (by decide) (fun | 0 => x | 1 => y | 2 => z | _ => weight [])

/-- The chart generator is produced in the existing original angular basis,
not a separately assumed coordinate-action convention. -/
theorem leftGenerator_original_basis (e : Q') (u : Triple) :
    PolyQuaternionCoeffEqv (leftGenerator e u) (originalLeftGenerator e u) :=
  ⟨basis_component 0 (2*e*u.x) (2*e*u.y) (2*e*u.z),
    basis_component 1 (2*e*u.x) (2*e*u.y) (2*e*u.z),
    basis_component 2 (2*e*u.x) (2*e*u.y) (2*e*u.z),
    basis_component 3 (2*e*u.x) (2*e*u.y) (2*e*u.z)⟩

/-- The exact cleared chart tangent on the SAME original left-generator basis.
The optional right chart and an arbitrary-source frame pullback are not supplied
by this leaf. No Haar/sphere equality is used on the chart coordinates. -/
theorem cayley_original_left_tangent (e : Q') (u : Triple) :
    PaddedQuaternionCoeffEqv 1 (clearedQuotientTangent e u)
      (qscale (cayleyD e) (PolyQuaternion.mul (originalLeftGenerator e u) (cayleyN e))) := by
  have h := (PaddedQuaternionCoeffEqv.of_coeff (leftGenerator_original_basis e u)).mul
    (PaddedQuaternionCoeffEqv.refl 1 (cayleyN e))
  exact (cayley_cleared_left_tangent e u).trans
    (paddedQuaternion_components
      ((PaddedPolynomialCoeffEqv.refl 1 (cayleyD e)).mul h.scalar)
      ((PaddedPolynomialCoeffEqv.refl 1 (cayleyD e)).mul h.iPart)
      ((PaddedPolynomialCoeffEqv.refl 1 (cayleyD e)).mul h.jPart)
      ((PaddedPolynomialCoeffEqv.refl 1 (cayleyD e)).mul h.kPart))
